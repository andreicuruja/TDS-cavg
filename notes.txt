fun main() {
    print("Quantos alunos há na sala? ")
    val quantidade = readLine()?.toIntOrNull()

    if (quantidade != null && quantidade > 0) {
        var somaNotas = 0.0

        for (i in 1..quantidade) {
            print("Digite a nota do aluno $i: ")
            val nota = readLine()?.toDoubleOrNull()

            if (nota != null && nota in 0.0..10.0) {
                somaNotas += nota
            } else {
                println("Nota invalida. Considerando 0.")
            }
        }

        val media = somaNotas / quantidade
        println("A media da turma é: %.2f".format(media))
    } else {
        println("Quantidade invalida de alunos.")
    }
}
